cmake_minimum_required(VERSION 3.16)
project(SinhSample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B4" CACHE STRING "Ascend SoC")
set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN installation")
set(RUN_MODE "npu" CACHE STRING "cpu, sim or npu")
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()
if(RUN_MODE STREQUAL "cpu")
    find_package(tikicpulib REQUIRED PATHS "${ASCEND_CANN_PACKAGE_PATH}/tools/tikicpulib/lib/cmake" NO_DEFAULT_PATH)
    add_executable(main main.cpp sinh_custom.cpp)
    target_link_libraries(main PRIVATE tikicpulib::${SOC_VERSION})
    target_compile_definitions(main PRIVATE ASCENDC_CPU_DEBUG _GLIBCXX_USE_CXX11_ABI=0)
else()
    if(RUN_MODE STREQUAL "sim")
        set(RUN_MODE "sim" CACHE STRING "Run mode" FORCE)
    elseif(NOT RUN_MODE STREQUAL "npu")
        message(FATAL_ERROR "RUN_MODE must be cpu, sim or npu")
    endif()
    include("${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
    ascendc_library(kernels STATIC sinh_custom.cpp)
    add_executable(main main.cpp)
    target_link_libraries(main PRIVATE kernels)
    if(RUN_MODE STREQUAL "sim")
        # CANN 8.5.2 model libraries omit DT_NEEDED entries; retain their link closure.
        target_link_libraries(main PRIVATE "-Wl,--no-as-needed" model_top stars ffts_model mcu_wrapper mcu_loop pem_davinci "-Wl,--as-needed")
    endif()
endif()

target_compile_definitions(main PRIVATE RUN_MODE_LABEL="${RUN_MODE}")
