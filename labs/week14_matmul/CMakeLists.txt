cmake_minimum_required(VERSION 3.16)
project(MatmulTilingLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOC_VERSION "Ascend910B4" CACHE STRING "SoC")
set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/ascend-toolkit/latest" CACHE PATH "CANN")
set(RUN_MODE "npu" CACHE STRING "mode")
set(CMAKE_BUILD_TYPE Debug CACHE STRING "type" FORCE)
include("${ASCEND_CANN_PACKAGE_PATH}/tools/tikcpp/ascendc_kernel_cmake/ascendc.cmake")
ascendc_library(kernels STATIC matmul_custom.cpp)
ascendc_compile_definitions(kernels PRIVATE HAVE_WORKSPACE HAVE_TILING)
add_executable(main main.cpp)
target_include_directories(main PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/include")
target_link_directories(main PRIVATE "${ASCEND_CANN_PACKAGE_PATH}/lib64")
target_link_libraries(main PRIVATE kernels tiling_api platform register ascendalog dl)
