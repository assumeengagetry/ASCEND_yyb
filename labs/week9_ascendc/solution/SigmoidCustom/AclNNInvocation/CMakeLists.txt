cmake_minimum_required(VERSION 3.16)
project(sigmoid_aclnn LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED ASCEND_CANN_PACKAGE_PATH)
    set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}")
endif()
if(NOT DEFINED OP_API_DIR)
    message(FATAL_ERROR "Pass -DOP_API_DIR=<install>/vendors/lab_sigmoid/op_api")
endif()
find_library(ASCENDCL_LIBRARY ascendcl PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64" NO_DEFAULT_PATH)
find_library(NNOPBASE_LIBRARY nnopbase PATHS "${ASCEND_CANN_PACKAGE_PATH}/lib64" NO_DEFAULT_PATH)
find_library(CUSTOM_OPAPI_LIBRARY cust_opapi PATHS "${OP_API_DIR}/lib" NO_DEFAULT_PATH)
if(NOT ASCENDCL_LIBRARY OR NOT NNOPBASE_LIBRARY OR NOT CUSTOM_OPAPI_LIBRARY)
    message(FATAL_ERROR "Missing ascendcl, nnopbase, or cust_opapi library; check CANN and OP_API_DIR")
endif()
add_executable(execute_sigmoid main.cpp)
target_include_directories(execute_sigmoid PRIVATE
    "${ASCEND_CANN_PACKAGE_PATH}/include"
    "${ASCEND_CANN_PACKAGE_PATH}/runtime/include"
    "${OP_API_DIR}/include")
target_link_libraries(execute_sigmoid PRIVATE "${CUSTOM_OPAPI_LIBRARY}" "${ASCENDCL_LIBRARY}" "${NNOPBASE_LIBRARY}")
set_target_properties(execute_sigmoid PROPERTIES BUILD_RPATH "${OP_API_DIR}/lib;${ASCEND_CANN_PACKAGE_PATH}/lib64")
